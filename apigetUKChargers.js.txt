// api/getUKChargers.js
export default async function handler(req, res) {
  try {
    // Extract query parameters and provide defaults
    let { lat, lon, distance } = req.query;
    lat = lat || 54.5;       // UK center default
    lon = lon || -3;
    distance = distance || 25;

    console.log("Request received with:", { lat, lon, distance });

    const apiKey = "9e32798c-5e2f-4146-bb18-5870fd74e0ec";

    const response = await fetch(
      `https://api.openchargemap.io/v3/poi/?output=json&countrycode=GB&latitude=${lat}&longitude=${lon}&maxresults=500&distance=${distance}&distanceunit=KM&key=${apiKey}`
    );

    if (!response.ok) {
      console.error("OpenChargeMap API returned status:", response.status);
      return res.status(response.status).json({ error: "OpenChargeMap API error" });
    }

    const data = await response.json();
    console.log("Data received from OpenChargeMap:", data.length, "entries");

    res.status(200).json(data);
  } catch (err) {
    console.error("Serverless function error:", err);
    res.status(500).json({ error: "Failed to fetch data from OpenChargeMap" });
  }
}
